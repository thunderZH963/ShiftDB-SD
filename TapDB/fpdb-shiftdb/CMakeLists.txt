cmake_minimum_required(VERSION 3.15)
project(fpdb-shiftdb LANGUAGES CXX)
option(SHIFTDB_SANITIZE "Enable address/undefined sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(fpdb-shiftdb src/planner.cpp src/resources.cpp src/state.cpp src/ownership.cpp src/handoff.cpp src/estimator.cpp)
target_include_directories(fpdb-shiftdb PUBLIC include)
target_compile_features(fpdb-shiftdb PUBLIC cxx_std_17)
target_link_libraries(fpdb-shiftdb PUBLIC Threads::Threads)
target_compile_options(fpdb-shiftdb PRIVATE -Wall -Wextra -Wpedantic)
if(SHIFTDB_SANITIZE)
 target_compile_options(fpdb-shiftdb PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(fpdb-shiftdb PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(shiftdb-tests tests/runtime_test.cpp)
target_link_libraries(shiftdb-tests PRIVATE fpdb-shiftdb)
foreach(suite planner resources state ownership handoff bridge)
 add_test(NAME shiftdb-${suite} COMMAND shiftdb-tests ${suite})
endforeach()
foreach(regression planner resources handoff estimator migration_gate)
 add_executable(shiftdb-${regression}-regressions tests/${regression}_regression_test.cpp)
 target_link_libraries(shiftdb-${regression}-regressions PRIVATE fpdb-shiftdb)
 add_test(NAME shiftdb-${regression}-regressions COMMAND shiftdb-${regression}-regressions)
endforeach()
add_executable(shiftdb-operator-continuation tests/operator_continuation_test.cpp)
target_link_libraries(shiftdb-operator-continuation PRIVATE fpdb-shiftdb)
add_test(NAME shiftdb-operator-continuation COMMAND shiftdb-operator-continuation)
add_executable(shiftdb-demo examples/handoff_demo.cpp)
target_link_libraries(shiftdb-demo PRIVATE fpdb-shiftdb)
add_test(NAME shiftdb-demo COMMAND shiftdb-demo)

option(SHIFTDB_ARROW_TESTS "Build Arrow state codecs against explicit Arrow installation" OFF)
if(SHIFTDB_ARROW_TESTS)
  if(NOT SHIFTDB_ARROW_INCLUDE_DIR OR NOT SHIFTDB_ARROW_LIBRARY)
    message(FATAL_ERROR "Set SHIFTDB_ARROW_INCLUDE_DIR and SHIFTDB_ARROW_LIBRARY")
  endif()
  add_library(shiftdb-arrow-state native/src/arrow_state.cpp)
  target_include_directories(shiftdb-arrow-state PUBLIC native/include "${SHIFTDB_ARROW_INCLUDE_DIR}")
  target_link_libraries(shiftdb-arrow-state PUBLIC fpdb-shiftdb "${SHIFTDB_ARROW_LIBRARY}")
  add_executable(shiftdb-arrow-tests native/tests/arrow_state_test.cpp)
  target_link_libraries(shiftdb-arrow-tests PRIVATE shiftdb-arrow-state)
  add_test(NAME shiftdb-arrow-state COMMAND shiftdb-arrow-tests)
endif()

# Compile the actual TapDB tuple appenders against an existing Arrow installation.
option(SHIFTDB_APPENDER_TESTS "Test non-destructive native TapDB appender snapshots" OFF)
if(SHIFTDB_APPENDER_TESTS)
  if(NOT SHIFTDB_ARROW_TESTS OR NOT SHIFTDB_EXPECTED_INCLUDE_DIR OR NOT SHIFTDB_SPDLOG_INCLUDE_DIR)
    message(FATAL_ERROR "Enable Arrow tests and supply expected/spdlog include directories")
  endif()
  add_executable(shiftdb-appender-tests native/tests/appender_state_test.cpp
    ../fpdb-tuple/src/ArrayAppender.cpp ../fpdb-tuple/src/ArrayAppenderWrapper.cpp)
  target_include_directories(shiftdb-appender-tests PRIVATE ../fpdb-tuple/include
    "${SHIFTDB_EXPECTED_INCLUDE_DIR}" "${SHIFTDB_SPDLOG_INCLUDE_DIR}")
  target_link_libraries(shiftdb-appender-tests PRIVATE shiftdb-arrow-state)
  add_test(NAME shiftdb-native-appender COMMAND shiftdb-appender-tests)
endif()
